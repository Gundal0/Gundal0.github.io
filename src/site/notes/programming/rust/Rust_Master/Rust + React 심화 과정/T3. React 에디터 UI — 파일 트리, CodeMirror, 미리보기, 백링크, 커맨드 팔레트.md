// src/store.ts
import { create } from "zustand";
import { api, NoteMeta } from "./lib/api";

export type Command = { id: string; name: string; source: string; run: () => void | Promise<void> };
type SaveCtx = { path: string; title: string; text: string };
type BeforeSave = (ctx: SaveCtx) => Promise<{ text: string; notices: string[] }>;

type State = {
  notes: NoteMeta[];
  current: NoteMeta | null;
  text: string;
  dirty: boolean;
  backlinks: string[];
  status: string;
  statusItems: Record<string, string>;   // 플러그인별 상태 표시줄 항목
  commands: Command[];
  beforeSave: BeforeSave | null;         // R4의 PluginHost가 설치 (순환 import 방지)

  openVault(path: string): Promise<void>;
  refreshNotes(): Promise<void>;
  openNote(n: NoteMeta): Promise<void>;
  openLink(title: string): Promise<void>;
  setText(t: string): void;
  save(): Promise<void>;
  notify(msg: string): void;
  registerCommand(c: Command): void;
  unregisterSource(source: string): void;
  setStatusItem(key: string, text: string | null): void;
  setBeforeSave(fn: BeforeSave | null): void;
};

export const useStore = create<State>((set, get) => ({
  notes: [], current: null, text: "", dirty: false, backlinks: [], status: "",
  statusItems: {}, commands: [], beforeSave: null,

  async openVault(path) {
    const notes = await api.openVault(path);
    set({ notes, status: `${notes.length}개 노트` });
  },
  async refreshNotes() { set({ notes: await api.listNotes() }); },

  async openNote(n) {
    await get().save();                                  // 떠나기 전 저장
    const [text, backlinks] = await Promise.all([api.read(n.path), api.backlinks(n.title)]);
    set({ current: n, text, dirty: false, backlinks, status: n.path });
  },

  async openLink(title) {
    const n = await api.openLink(title);                 // 없으면 Rust가 새로 만듦
    await get().refreshNotes();
    await get().openNote(n);
  },

  setText(text) { set({ text, dirty: true }); },

  async save() {
    const { current, dirty, beforeSave } = get();
    if (!current || !dirty) return;
    let { text } = get();
    let notices: string[] = [];
    if (beforeSave) ({ text, notices } = await beforeSave({ path: current.path, title: current.title, text }));
    await api.save(current.path, text);
    set({ text, dirty: false, backlinks: await api.backlinks(current.title),
          status: notices.length ? `💾 ${notices.join(" · ")}` : `💾 ${current.title} 저장됨` });
  },

  notify(msg) { set({ status: msg }); },
  registerCommand(c) { set((s) => ({ commands: [...s.commands.filter((x) => x.id !== c.id), c] })); },
  unregisterSource(source) { set((s) => ({ commands: s.commands.filter((c) => c.source !== source) })); },
  setStatusItem(key, text) {
    set((s) => {
      const next = { ...s.statusItems };
      if (text === null) delete next[key]; else next[key] = text;
      return { statusItems: next };
    });
  },
  setBeforeSave(fn) { set({ beforeSave: fn }); },
}));
```

// src/lib/wikilink.ts
export const WIKILINK = /\[\[([^\]|#]+)(#[^\]|]*)?(?:\|([^\]]*))?\]\]/g;

/** [[대상|별칭]] → [별칭](#wiki:대상)  (react-markdown이 안전하게 렌더링할 수 있는 일반 링크로) */
export function wikilinksToMarkdown(text: string): string {
  return text.replace(WIKILINK, (_m, target: string, _h, alias?: string) => {
    const label = (alias ?? target).trim().replace(/[[\]]/g, "");
    return `[${label}](#wiki:${encodeURIComponent(target.trim())})`;
  });
}

export function headings(text: string): { level: number; title: string }[] {
  let inCode = false;
  const out: { level: number; title: string }[] = [];
  for (const line of text.split("\n")) {
    if (line.trimStart().startsWith("```")) { inCode = !inCode; continue; }
    const m = !inCode && /^(#{1,6})\s+(.+)$/.exec(line);
    if (m) out.push({ level: m[1].length, title: m[2].trim() });
  }
  return out;
}

// src/components/Editor.tsx
import { useEffect, useMemo } from "react";
import CodeMirror from "@uiw/react-codemirror";
import { markdown } from "@codemirror/lang-markdown";
import { autocompletion, CompletionContext } from "@codemirror/autocomplete";
import { EditorView, keymap } from "@codemirror/view";
import { useStore } from "../store";
import { editorBridge } from "../lib/editorBridge";

// [[ 뒤에서 노트 제목 자동완성 (최신 notes는 getState()로 읽어 확장 재생성 방지)
function wikiCompletion(ctx: CompletionContext) {
  const m = ctx.matchBefore(/\[\[[^\]|#]*/);
  if (!m) return null;
  return {
    from: m.from + 2,
    options: useStore.getState().notes.map((n) => ({ label: n.title, detail: n.path, apply: `${n.title}]]` })),
  };
}

export function Editor() {
  const text = useStore((s) => s.text);
  const current = useStore((s) => s.current);
  const dirty = useStore((s) => s.dirty);
  const setText = useStore((s) => s.setText);

  // 확장은 한 번만 만든다 (매 렌더마다 만들면 에디터가 재설정됨)
  const extensions = useMemo(() => [
    markdown(),
    EditorView.lineWrapping,
    autocompletion({ override: [wikiCompletion] }),
    keymap.of([{ key: "Mod-s", run: () => { void useStore.getState().save(); return true; } }]),
  ], []);

  // 입력이 1.5초 멈추면 자동 저장
  useEffect(() => {
    if (!dirty) return;
    const t = setTimeout(() => void useStore.getState().save(), 1500);
    return () => clearTimeout(t);
  }, [text, dirty]);

  if (!current) return <section className="empty">🦀 노트를 선택하거나 Ctrl/⌘+O로 여세요</section>;

  return (
    <section className="editor">
      <CodeMirror
        value={text}
        height="100%"
        extensions={extensions}
        onChange={setText}
        onCreateEditor={(view) => editorBridge.attach(view)}
      />
    </section>
  );
}

// src/components/CommandPalette.tsx
import { useEffect, useMemo, useState } from "react";
import { useStore } from "../store";
import { api } from "../lib/api";

type Item = { key: string; label: string; hint?: string; run: () => void | Promise<void> };

export function CommandPalette({ mode, onClose }: { mode: "notes" | "commands"; onClose: () => void }) {
  const { notes, commands, openNote } = useStore();
  const [q, setQ] = useState("");
  const [sel, setSel] = useState(0);
  const [hits, setHits] = useState<Item[]>([]);

  // "/키워드" → Rust 전문 검색
  useEffect(() => {
    if (mode !== "notes" || !q.startsWith("/")) { setHits([]); return; }
    const t = setTimeout(async () => {
      const res = await api.search(q.slice(1));
      setHits(res.map((h) => ({ key: `${h.path}:${h.line}`, label: `${h.title}:${h.line}`, hint: h.snippet,
        run: () => { const n = notes.find((x) => x.path === h.path); if (n) return openNote(n); } })));
    }, 200);
    return () => clearTimeout(t);
  }, [q, mode, notes, openNote]);

  const items: Item[] = useMemo(() => {
    const lq = q.toLowerCase();
    if (mode === "commands")
      return commands.filter((c) => c.name.toLowerCase().includes(lq))
        .map((c) => ({ key: c.id, label: c.name, hint: c.source === "core" ? undefined : `🧩 ${c.source}`, run: c.run }));
    if (q.startsWith("/")) return hits;
    return notes.filter((n) => n.title.toLowerCase().includes(lq)).slice(0, 50)
      .map((n) => ({ key: n.path, label: n.title, hint: n.path, run: () => openNote(n) }));
  }, [q, mode, commands, notes, hits, openNote]);

  async function choose(i: number) { onClose(); await items[i]?.run(); }

  return (
    <div className="palette-backdrop" onClick={onClose}>
      <div className="palette" onClick={(e) => e.stopPropagation()}>
        <input autoFocus value={q}
          placeholder={mode === "commands" ? "명령 검색" : "노트 이름 또는 /전문검색"}
          onChange={(e) => { setQ(e.target.value); setSel(0); }}
          onKeyDown={(e) => {
            if (e.key === "ArrowDown") setSel((s) => Math.min(s + 1, items.length - 1));
            if (e.key === "ArrowUp") setSel((s) => Math.max(s - 1, 0));
            if (e.key === "Enter") void choose(sel);
            if (e.key === "Escape") onClose();
          }} />
        <ul>
          {items.map((it, i) => (
            <li key={it.key} className={i === sel ? "active" : ""} onClick={() => void choose(i)}>
              {it.label} {it.hint && <small>{it.hint}</small>}
            </li>
          ))}
        </ul>
      </div>
    </div>
  );
}

export default function App() {

  const status = useStore((s) => s.status);
  const statusItems = useStore((s) => s.statusItems);

  useEffect(() => {
    const s = useStore.getState();
    // 기본 명령 (플러그인 명령과 같은 레지스트리)

      await useStore.getState().openNote(meta);
      useStore.getState().notify(`✏️ ${n}개 노트의 링크 갱신`);
    } });
    s.registerCommand({ id: "core:reload-plugins", name: "플러그인 다시 불러오기", source: "core", run: () => pluginHost.reload() });

    void pluginHost.loadAll();                              // R4

    const onKey = (e: KeyboardEvent) => {
      const mod = e.ctrlKey || e.metaKey;                   // Windows/Linux Ctrl, macOS Cmd
      if (mod && e.key.toLowerCase() === "o") { e.preventDefault(); setPalette("notes"); }
      if (mod && e.key.toLowerCase() === "p") { e.preventDefault(); setPalette("commands"); }
    };
    window.addEventListener("keydown", onKey);

    // 외부 변경 → 300ms 디바운스 후 목록 갱신
    let t: ReturnType<typeof setTimeout>;
    const unlisten = listen("vault-changed", () => {
      clearTimeout(t);
      t = setTimeout(() => void useStore.getState().refreshNotes().catch(() => {}), 300);
    });
    return () => { window.removeEventListener("keydown", onKey); void unlisten.then((f) => f()); };
  }, []);

  return (
    <div className="app">
      <div className="layout">
        <FileTree />
        <Editor />
        <Preview />
        <SidePanel />
      </div>
      <footer className="statusbar">
        <span>{status}</span>
        {Object.entries(statusItems).map(([k, v]) => <span key={k} className="item">{v}</span>)}
      </footer>
      {palette && <CommandPalette mode={palette} onClose={() => setPalette(null)} />}
    </div>
  );
}

// src/lib/wikilink.ts
export const WIKILINK = /\[\[([^\]|#]+)(#[^\]|]*)?(?:\|([^\]]*))?\]\]/g;

/** [[대상|별칭]] → [별칭](#wiki:대상)  (react-markdown이 안전하게 렌더링할 수 있는 일반 링크로) */
export function wikilinksToMarkdown(text: string): string {
  return text.replace(WIKILINK, (_m, target: string, _h, alias?: string) => {
    const label = (alias ?? target).trim().replace(/[[\]]/g, "");
    return `[${label}](#wiki:${encodeURIComponent(target.trim())})`;
  });
}

export function headings(text: string): { level: number; title: string }[] {
  let inCode = false;
  const out: { level: number; title: string }[] = [];
  for (const line of text.split("\n")) {
    if (line.trimStart().startsWith("```")) { inCode = !inCode; continue; }
    const m = !inCode && /^(#{1,6})\s+(.+)$/.exec(line);
    if (m) out.push({ level: m[1].length, title: m[2].trim() });
  }
  return out;
}

// src/components/Editor.tsx
import { useEffect, useMemo } from "react";
import CodeMirror from "@uiw/react-codemirror";
import { markdown } from "@codemirror/lang-markdown";
import { autocompletion, CompletionContext } from "@codemirror/autocomplete";
import { EditorView, keymap } from "@codemirror/view";
import { useStore } from "../store";
import { editorBridge } from "../lib/editorBridge";

// [[ 뒤에서 노트 제목 자동완성 (최신 notes는 getState()로 읽어 확장 재생성 방지)
function wikiCompletion(ctx: CompletionContext) {
  const m = ctx.matchBefore(/\[\[[^\]|#]*/);
  if (!m) return null;
  return {
    from: m.from + 2,
    options: useStore.getState().notes.map((n) => ({ label: n.title, detail: n.path, apply: `${n.title}]]` })),
  };
}

export function Editor() {
  const text = useStore((s) => s.text);
  const current = useStore((s) => s.current);
  const dirty = useStore((s) => s.dirty);
  const setText = useStore((s) => s.setText);

  // 확장은 한 번만 만든다 (매 렌더마다 만들면 에디터가 재설정됨)
  const extensions = useMemo(() => [
    markdown(),
    EditorView.lineWrapping,
    autocompletion({ override: [wikiCompletion] }),
    keymap.of([{ key: "Mod-s", run: () => { void useStore.getState().save(); return true; } }]),
  ], []);

  // 입력이 1.5초 멈추면 자동 저장
  useEffect(() => {
    if (!dirty) return;
    const t = setTimeout(() => void useStore.getState().save(), 1500);
    return () => clearTimeout(t);
  }, [text, dirty]);

  if (!current) return <section className="empty">🦀 노트를 선택하거나 Ctrl/⌘+O로 여세요</section>;

  return (
    <section className="editor">
      <CodeMirror
        value={text}
        height="100%"
        extensions={extensions}
        onChange={setText}
        onCreateEditor={(view) => editorBridge.attach(view)}
      />
    </section>
  );
}

// src/components/CommandPalette.tsx
import { useEffect, useMemo, useState } from "react";
import { useStore } from "../store";
import { api } from "../lib/api";

type Item = { key: string; label: string; hint?: string; run: () => void | Promise<void> };

export function CommandPalette({ mode, onClose }: { mode: "notes" | "commands"; onClose: () => void }) {
  const { notes, commands, openNote } = useStore();
  const [q, setQ] = useState("");
  const [sel, setSel] = useState(0);
  const [hits, setHits] = useState<Item[]>([]);

  // "/키워드" → Rust 전문 검색
  useEffect(() => {
    if (mode !== "notes" || !q.startsWith("/")) { setHits([]); return; }
    const t = setTimeout(async () => {
      const res = await api.search(q.slice(1));
      setHits(res.map((h) => ({ key: `${h.path}:${h.line}`, label: `${h.title}:${h.line}`, hint: h.snippet,
        run: () => { const n = notes.find((x) => x.path === h.path); if (n) return openNote(n); } })));
    }, 200);
    return () => clearTimeout(t);
  }, [q, mode, notes, openNote]);

  const items: Item[] = useMemo(() => {
    const lq = q.toLowerCase();
    if (mode === "commands")
      return commands.filter((c) => c.name.toLowerCase().includes(lq))
        .map((c) => ({ key: c.id, label: c.name, hint: c.source === "core" ? undefined : `🧩 ${c.source}`, run: c.run }));
    if (q.startsWith("/")) return hits;
    return notes.filter((n) => n.title.toLowerCase().includes(lq)).slice(0, 50)
      .map((n) => ({ key: n.path, label: n.title, hint: n.path, run: () => openNote(n) }));
  }, [q, mode, commands, notes, hits, openNote]);

  async function choose(i: number) { onClose(); await items[i]?.run(); }

  return (
    <div className="palette-backdrop" onClick={onClose}>
      <div className="palette" onClick={(e) => e.stopPropagation()}>
        <input autoFocus value={q}
          placeholder={mode === "commands" ? "명령 검색" : "노트 이름 또는 /전문검색"}
          onChange={(e) => { setQ(e.target.value); setSel(0); }}
          onKeyDown={(e) => {
            if (e.key === "ArrowDown") setSel((s) => Math.min(s + 1, items.length - 1));
            if (e.key === "ArrowUp") setSel((s) => Math.max(s - 1, 0));
            if (e.key === "Enter") void choose(sel);
            if (e.key === "Escape") onClose();
          }} />
        <ul>
          {items.map((it, i) => (
            <li key={it.key} className={i === sel ? "active" : ""} onClick={() => void choose(i)}>
              {it.label} {it.hint && <small>{it.hint}</small>}
            </li>
          ))}
        </ul>
      </div>
    </div>
  );
}

import { useEffect, useState } from "react";
import { listen } from "@tauri-apps/api/event";
import { useStore } from "./store";
import { api } from "./lib/api";
import { FileTree } from "./components/FileTree";
import { Editor } from "./components/Editor";
import { Preview } from "./components/Preview";
import { SidePanel } from "./components/SidePanel";
import { CommandPalette } from "./components/CommandPalette";
import { pluginHost } from "./plugins/PluginHost";
import "./App.css";

export default function App() {
  const [palette, setPalette] = useState<null | "notes" | "commands">(null);
  const status = useStore((s) => s.status);
  const statusItems = useStore((s) => s.statusItems);

  useEffect(() => {
    const s = useStore.getState();
    // 기본 명령 (플러그인 명령과 같은 레지스트리)
    s.registerCommand({ id: "core:rename", name: "노트 이름 바꾸기", source: "core", run: async () => {
      const cur = useStore.getState().current; if (!cur) return;
      const name = prompt("새 이름", cur.title); if (!name) return;
      const [meta, n] = await api.rename(cur.path, name);
      await useStore.getState().refreshNotes();
      await useStore.getState().openNote(meta);
      useStore.getState().notify(`✏️ ${n}개 노트의 링크 갱신`);
    } });
    s.registerCommand({ id: "core:reload-plugins", name: "플러그인 다시 불러오기", source: "core", run: () => pluginHost.reload() });

    void pluginHost.loadAll();                              // R4

    const onKey = (e: KeyboardEvent) => {
      const mod = e.ctrlKey || e.metaKey;                   // Windows/Linux Ctrl, macOS Cmd
      if (mod && e.key.toLowerCase() === "o") { e.preventDefault(); setPalette("notes"); }
      if (mod && e.key.toLowerCase() === "p") { e.preventDefault(); setPalette("commands"); }
    };
    window.addEventListener("keydown", onKey);

    // 외부 변경 → 300ms 디바운스 후 목록 갱신
    let t: ReturnType<typeof setTimeout>;
    const unlisten = listen("vault-changed", () => {
      clearTimeout(t);
      t = setTimeout(() => void useStore.getState().refreshNotes().catch(() => {}), 300);
    });
    return () => { window.removeEventListener("keydown", onKey); void unlisten.then((f) => f()); };
  }, []);

  return (
    <div className="app">
      <div className="layout">
        <FileTree />
        <Editor />
        <Preview />
        <SidePanel />
      </div>
      <footer className="statusbar">
        <span>{status}</span>
        {Object.entries(statusItems).map(([k, v]) => <span key={k} className="item">{v}</span>)}
      </footer>
      {palette && <CommandPalette mode={palette} onClose={() => setPalette(null)} />}
    </div>
  );
}
