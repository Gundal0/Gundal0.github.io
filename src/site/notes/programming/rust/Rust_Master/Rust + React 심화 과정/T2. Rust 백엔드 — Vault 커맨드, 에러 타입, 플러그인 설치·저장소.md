## 🧩 문제 1 — 프론트엔드가 에러 종류를 구분할 수 있어야 한다

## 💡 해결 1 — 직렬화 가능한 에러 열거형

```rust
// src-tauri/src/error.rs
use serde::{ser::SerializeStruct, Serialize, Serializer};

#[derive(Debug, thiserror::Error)]
pub enum AppError {
    #[error(transparent)]
    Core(#[from] crabnote_core::CoreError),
    #[error("파일 오류: {0}")]
    Io(#[from] std::io::Error),
    #[error("zip 오류: {0}")]
    Zip(#[from] zip::result::ZipError),
    #[error("JSON 오류: {0}")]
    Json(#[from] serde_json::Error),
    #[error("Vault가 열려 있지 않습니다")]
    NoVault,
    #[error("잘못된 플러그인: {0}")]
    InvalidPlugin(String),
}

impl AppError {
    fn kind(&self) -> &'static str {
        match self {
            AppError::Core(_) => "core",
            AppError::Io(_) => "io",
            AppError::Zip(_) | AppError::InvalidPlugin(_) => "plugin",
            AppError::Json(_) => "json",
            AppError::NoVault => "no_vault",
        }
    }
}

// JS에서는 { kind: "no_vault", message: "..." } 로 받는다
impl Serialize for AppError {
    fn serialize<S: Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
        let mut st = s.serialize_struct("AppError", 2)?;
        st.serialize_field("kind", self.kind())?;
        st.serialize_field("message", &self.to_string())?;
        st.end()
    }
}

pub type AppResult<T> = Result<T, AppError>;
```

```tsx
// 프론트엔드에서
try { await api.read("a.md"); }
catch (e) {
  const err = e as { kind: string; message: string };
  if (err.kind === "no_vault") openVaultDialog(); else notify(err.message);
}
```

## 🧩 문제 2 — Vault 기능을 커맨드로 노출해야 한다

## 💡 해결 2 — `vault_cmds.rs`

```rust
// src-tauri/src/vault_cmds.rs
use crate::error::{AppError, AppResult};
use crabnote_core::{links::LinkIndex, search, NoteMeta, Vault};
use std::sync::Mutex;
use tauri::{AppHandle, Emitter, State};

#[derive(Default)]
pub struct VaultState {
    vault: Mutex<Option<Vault>>,
    index: Mutex<LinkIndex>,
    watcher: Mutex<Option<notify::RecommendedWatcher>>,
}

fn with_vault<T>(s: &VaultState, f: impl FnOnce(&mut Vault) -> crabnote_core::Result<T>) -> AppResult<T> {
    let mut g = s.vault.lock().unwrap();
    Ok(f(g.as_mut().ok_or(AppError::NoVault)?)?)
}

fn reindex(s: &VaultState) {
    if let Some(v) = s.vault.lock().unwrap().as_ref() {
        let pairs = v.notes().filter_map(|n| v.read(&n.path).ok().map(|b| (n.path.as_str(), b)));
        *s.index.lock().unwrap() = LinkIndex::build(pairs);
    }
}

#[tauri::command]
pub fn open_vault(app: AppHandle, path: String, state: State<'_, VaultState>) -> AppResult<Vec<NoteMeta>> {
    use notify::Watcher;
    let vault = Vault::open(&path)?;
    let h = app.clone();
    let mut w = notify::recommended_watcher(move |r: notify::Result<notify::Event>| {
        if r.is_ok() { let _ = h.emit("vault-changed", ()); }
    }).map_err(|e| AppError::InvalidPlugin(e.to_string()))?;
    let _ = w.watch(vault.root(), notify::RecursiveMode::Recursive);
    let notes = vault.notes().cloned().collect();
    *state.watcher.lock().unwrap() = Some(w);
    *state.vault.lock().unwrap() = Some(vault);
    reindex(&state);
    Ok(notes)
}

#[tauri::command]
pub fn list_notes(state: State<'_, VaultState>) -> AppResult<Vec<NoteMeta>> {
    with_vault(&state, |v| { v.rescan(); Ok(v.notes().cloned().collect()) })
}

#[tauri::command]
pub fn read_note(path: String, state: State<'_, VaultState>) -> AppResult<String> {
    with_vault(&state, |v| v.read(&path))
}

#[tauri::command]
pub fn save_note(path: String, text: String, state: State<'_, VaultState>) -> AppResult<()> {
    with_vault(&state, |v| v.write(&path, &text))?;
    reindex(&state);
    Ok(())
}

#[tauri::command]
pub fn open_link(title: String, state: State<'_, VaultState>) -> AppResult<NoteMeta> {
    let n = with_vault(&state, |v| match v.find_by_title(&title) {
        Some(n) => Ok(n.clone()),
        None => v.create(&title),
    })?;
    reindex(&state);
    Ok(n)
}

#[tauri::command]
pub fn rename_note(path: String, new_title: String, state: State<'_, VaultState>) -> AppResult<(NoteMeta, usize)> {
    let r = with_vault(&state, |v| v.rename(&path, &new_title))?;
    reindex(&state);
    Ok(r)
}

#[tauri::command]
pub fn backlinks(title: String, state: State<'_, VaultState>) -> Vec<String> {
    state.index.lock().unwrap().backlinks(&title)
}

#[tauri::command]
pub fn search_notes(query: String, state: State<'_, VaultState>) -> AppResult<Vec<search::Hit>> {
    with_vault(&state, |v| Ok(search::search(v, &query, 200)))
}
```

<aside>  
💡

`crabnote-core` 의 `Vault`, `LinkIndex`, `search` 는 심화 과정 A의 T2 코드와 동일합니다. 마크다운 렌더링은 이번에는 React 쪽(react-markdown)에서 하므로 `render` 모듈은 쓰지 않습니다. 하나의 코어를 두 제품이 공유하는 것이 워크스페이스 분리의 힘입니다.

</aside>

## 🧩 문제 3 — 플러그인을 안전하게 설치하고, 플러그인별 데이터를 저장해야 한다

## 💡 해결 3 — `plugin_cmds.rs` (체크섬 검증 + Zip Slip 방어)

```rust
// src-tauri/src/plugin_cmds.rs
use crate::error::{AppError, AppResult};
use serde::{Deserialize, Serialize};
use sha2::{Digest, Sha256};
use std::collections::BTreeSet;
use std::io::Cursor;
use std::path::{Path, PathBuf};
use tauri::{AppHandle, Manager};

const MAX_PLUGIN_BYTES: u64 = 5 * 1024 * 1024;  // main.js 5MB 제한

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct JsPluginManifest {
    pub id: String,
    pub name: String,
    pub version: String,
    #[serde(default)] pub author: String,
    #[serde(default)] pub description: String,
    #[serde(default)] pub min_app_version: String,
    #[serde(default)] pub permissions: Vec<String>,
    #[serde(default = "default_main")] pub main: String,
}
fn default_main() -> String { "main.js".into() }

#[derive(Serialize)]
pub struct JsPlugin { manifest: JsPluginManifest, code: String, enabled: bool }

fn plugins_dir(app: &AppHandle) -> AppResult<PathBuf> {
    let d = app.path().app_data_dir().map_err(|e| AppError::InvalidPlugin(e.to_string()))?.join("plugins");
    std::fs::create_dir_all(&d)?;
    Ok(d)
}

fn validate(m: &JsPluginManifest) -> AppResult<()> {
    let id_ok = !m.id.is_empty() && m.id.len() <= 64
        && m.id.chars().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-');
    let main_ok = m.main.ends_with(".js") && !m.main.contains(['/', '\\']) && !m.main.contains("..");
    if !id_ok { return Err(AppError::InvalidPlugin(format!("id 형식 오류: {}", m.id))); }
    if !main_ok { return Err(AppError::InvalidPlugin(format!("main 형식 오류: {}", m.main))); }
    Ok(())
}

fn read_disabled(dir: &Path) -> BTreeSet<String> {
    std::fs::read_to_string(dir.join("disabled.json")).ok()
        .and_then(|s| serde_json::from_str(&s).ok()).unwrap_or_default()
}

#[tauri::command]
pub fn js_plugins_dir(app: AppHandle) -> AppResult<String> {
    Ok(plugins_dir(&app)?.display().to_string())
}

#[tauri::command]
pub fn list_js_plugins(app: AppHandle) -> AppResult<Vec<JsPlugin>> {
    let dir = plugins_dir(&app)?;
    let disabled = read_disabled(&dir);
    let mut out = vec![];
    for entry in std::fs::read_dir(&dir)?.flatten().filter(|e| e.path().is_dir()) {
        let p = entry.path();
        let Ok(raw) = std::fs::read_to_string(p.join("manifest.json")) else { continue };
        let Ok(manifest) = serde_json::from_str::<JsPluginManifest>(&raw) else { continue };
        if validate(&manifest).is_err() { continue; }
        let main = p.join(&manifest.main);
        if std::fs::metadata(&main).map(|m| m.len() > MAX_PLUGIN_BYTES).unwrap_or(true) { continue; }
        let code = std::fs::read_to_string(&main)?;
        let enabled = !disabled.contains(&manifest.id);
        out.push(JsPlugin { manifest, code, enabled });
    }
    out.sort_by(|a, b| a.manifest.name.cmp(&b.manifest.name));
    Ok(out)
}

#[tauri::command]
pub fn set_js_plugin_enabled(app: AppHandle, id: String, enabled: bool) -> AppResult<()> {
    let dir = plugins_dir(&app)?;
    let mut disabled = read_disabled(&dir);
    if enabled { disabled.remove(&id); } else { disabled.insert(id); }
    std::fs::write(dir.join("disabled.json"), serde_json::to_string(&disabled)?)?;
    Ok(())
}

/// zip(manifest.json + main.js)을 설치. sha256이 주어지면 먼저 무결성 검증.
#[tauri::command]
pub fn install_js_plugin(app: AppHandle, zip_path: String, sha256: Option<String>) -> AppResult<JsPluginManifest> {
    let bytes = std::fs::read(&zip_path)?;
    if let Some(expected) = sha256 {
        let actual = format!("{:x}", Sha256::digest(&bytes));
        if !actual.eq_ignore_ascii_case(expected.trim()) {
            return Err(AppError::InvalidPlugin(format!("체크섬 불일치 (기대 {expected}, 실제 {actual})")));
        }
    }

    let mut archive = zip::ZipArchive::new(Cursor::new(bytes))?;
    let manifest: JsPluginManifest = serde_json::from_reader(archive.by_name("manifest.json")?)?;
    validate(&manifest)?;

    let dest = plugins_dir(&app)?.join(&manifest.id);
    let staging = dest.with_extension("installing");
    let _ = std::fs::remove_dir_all(&staging);
    std::fs::create_dir_all(&staging)?;

    for i in 0..archive.len() {
        let mut file = archive.by_index(i)?;
        // 🔒 Zip Slip 방어: "../" 나 절대 경로를 가진 항목은 enclosed_name()이 None을 돌려준다
        let Some(rel) = file.enclosed_name() else {
            return Err(AppError::InvalidPlugin(format!("위험한 경로: {}", file.name())));
        };
        if file.size() > MAX_PLUGIN_BYTES {
            return Err(AppError::InvalidPlugin("파일이 너무 큽니다".into()));
        }
        let out = staging.join(rel);
        if file.is_dir() { std::fs::create_dir_all(&out)?; continue; }
        if let Some(parent) = out.parent() { std::fs::create_dir_all(parent)?; }
        std::io::copy(&mut file, &mut std::fs::File::create(&out)?)?;
    }

    // 스테이징 폴더에 다 푼 뒤 교체 → 설치 도중 실패해도 기존 플러그인이 깨지지 않음
    let _ = std::fs::remove_dir_all(&dest);
    std::fs::rename(&staging, &dest)?;
    Ok(manifest)
}

fn storage_file(app: &AppHandle, id: &str) -> AppResult<PathBuf> {
    if !id.chars().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-') {
        return Err(AppError::InvalidPlugin(id.into()));
    }
    let d = app.path().app_data_dir().map_err(|e| AppError::InvalidPlugin(e.to_string()))?.join("plugin-data");
    std::fs::create_dir_all(&d)?;
    Ok(d.join(format!("{id}.json")))
}

#[tauri::command]
pub fn plugin_storage_get(app: AppHandle, id: String) -> AppResult<serde_json::Value> {
    let f = storage_file(&app, &id)?;
    Ok(std::fs::read_to_string(f).ok().and_then(|s| serde_json::from_str(&s).ok()).unwrap_or(serde_json::Value::Null))
}

#[tauri::command]
pub fn plugin_storage_set(app: AppHandle, id: String, value: serde_json::Value) -> AppResult<()> {
    let s = serde_json::to_string(&value)?;
    if s.len() > 1024 * 1024 { return Err(AppError::InvalidPlugin("저장 데이터는 1MB 이하".into())); }
    std::fs::write(storage_file(&app, &id)?, s)?;
    Ok(())
}
```

// src-tauri/src/lib.rs
mod error;

    tauri::Builder::default()
        .plugin(tauri_plugin_dialog::init())

        .manage(vault_cmds::VaultState::default())
        .invoke_handler(tauri::generate_handler![
            vault_cmds::open_vault, vault_cmds::list_notes, vault_cmds::read_note, vault_cmds::save_note,
            vault_cmds::open_link, vault_cmds::rename_note, vault_cmds::backlinks, vault_cmds::search_notes,
            plugin_cmds::list_js_plugins, plugin_cmds::set_js_plugin_enabled, plugin_cmds::install_js_plugin,
            plugin_cmds::plugin_storage_get, plugin_cmds::plugin_storage_set, plugin_cmds::js_plugins_dir,
        ])
        .run(tauri::generate_context!())
        .expect("CrabNote React 실행 실패");
}
