```tsx
// src/lib/wikilink.test.ts
import { describe, expect, it } from "vitest";
import { headings, wikilinksToMarkdown } from "./wikilink";

describe("wikilinksToMarkdown", () => {
  it("별칭·제목 앵커를 처리한다", () => {
    expect(wikilinksToMarkdown("[[Rust 언어|러스트]] [[A#제목]]"))
      .toBe("[러스트](#wiki:Rust%20%EC%96%B8%EC%96%B4) [A](#wiki:A)");
  });
  it("대괄호가 들어간 별칭으로 링크를 깨지 않는다", () => {
    expect(wikilinksToMarkdown("[[A|x[y]]]")).not.toContain("[x[y]");
  });
});

describe("headings", () => {
  it("코드 블록 안의 #은 무시한다", () => {
    expect(headings("# A\n```\n# not\n```\n## B").map((h) => h.title)).toEqual(["A", "B"]);
  });
});
```

```rust
// src-tauri/src/plugin_cmds.rs 하단 — Zip Slip 방어 테스트 (설치 로직을 AppHandle 없는 함수로 분리해 두면 테스트가 쉬움)
#[cfg(test)]
mod tests {
    use std::io::Write;

    fn zip_with(entries: &[(&str, &str)]) -> Vec<u8> {
        let mut buf = std::io::Cursor::new(Vec::new());
        let mut z = zip::ZipWriter::new(&mut buf);
        for (name, body) in entries {
            z.start_file(*name, zip::write::SimpleFileOptions::default()).unwrap();
            z.write_all(body.as_bytes()).unwrap();
        }
        z.finish().unwrap();
        buf.into_inner()
    }

    #[test]
    fn enclosed_name_rejects_traversal() {
        let bytes = zip_with(&[("manifest.json", "{}"), ("../evil.txt", "x")]);
        let mut a = zip::ZipArchive::new(std::io::Cursor::new(bytes)).unwrap();
        assert!(a.by_name("../evil.txt").unwrap().enclosed_name().is_none());
        assert!(a.by_name("manifest.json").unwrap().enclosed_name().is_some());
    }
}
```

## 🧩 문제 2 — 플러그인 샌드박스가 실제로 막는지 확인해야 한다

## 💡 해결 2 — "악성 플러그인" 테스트 세트

개발 빌드에서 아래 플러그인들을 설치해 **모두 실패하는지** 확인합니다.

|시도|플러그인 코드|기대 결과|
|---|---|---|
|DOM 접근|`document.body.innerHTML = ""`|워커에 `document` 없음 → 오류 알림|
|외부 전송|`fetch("<https://example.com>", {method:"POST"})`|CSP `connect-src` 차단|
|외부 스크립트|`importScripts("<https://cdn.example/x.js>")`|CSP `script-src` 차단|
|권한 없는 API|(permissions: []) `await api.vault.read("a.md")`|`권한 없음: vault:read`|
|숨겨진 API|`postMessage({type:"rpc", id:1, method:"fs.delete"})`|`알 수 없는 API`|
|무한 루프|`api.onSave(() => { while (true) {} })`|2초 후 플러그인 종료, 저장은 성공|
|Vault 탈출|`await api.vault.read("../../.ssh/id_rsa")` (vault:read 보유)|Rust `OutsideVault` 거부|

```jsx
// 예: 무한 루프 플러그인 main.js
crabnote.onload((api) => {
  api.onSave(() => { while (true) {} });   // 호스트가 2초 후 terminate() 해야 함
});
```

<aside>  
💡

이 표를 보안 회귀 테스트로 보관하세요. E2E 자동화는 `tauri-driver`(WebDriver)로 가능하지만 현재 Windows·Linux만 지원하므로, macOS는 수동 점검을 병행합니다.

</aside>

## 🧩 문제 3 — 세 OS용 설치 파일을 자동으로 만들어야 한다

## 💡 해결 3 — GitHub Actions (React 빌드 포함)

```yaml
# .github/workflows/release.yml
name: release
on:
  push:
    tags: ["v*"]
jobs:
  test:
    strategy:
      matrix: { os: [ubuntu-22.04, macos-latest, windows-latest] }
    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: lts/* }
      - uses: dtolnay/rust-toolchain@stable
      - if: matrix.os == 'ubuntu-22.04'
        run: sudo apt-get update && sudo apt-get install -y libwebkit2gtk-4.1-dev libappindicator3-dev librsvg2-dev patchelf
      - run: npm ci && npm test
      - run: cargo test --workspace

  build:
    needs: test
    permissions: { contents: write }
    strategy:
      fail-fast: false
      matrix:
        include:
          - { platform: macos-latest, args: "--target universal-apple-darwin" }   # Intel + Apple Silicon 하나로
          - { platform: ubuntu-22.04, args: "" }
          - { platform: windows-latest, args: "" }
    runs-on: ${{ matrix.platform }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: lts/* }
      - uses: dtolnay/rust-toolchain@stable
        with:
          targets: ${{ matrix.platform == 'macos-latest' && 'aarch64-apple-darwin,x86_64-apple-darwin' || '' }}
      - if: matrix.platform == 'ubuntu-22.04'
        run: sudo apt-get update && sudo apt-get install -y libwebkit2gtk-4.1-dev libappindicator3-dev librsvg2-dev patchelf
      - run: npm ci                       # tauri build 가 beforeBuildCommand(npm run build)로 React 빌드
      - uses: tauri-apps/tauri-action@v0
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        with:
          tagName: v__VERSION__
          releaseName: "CrabNote React v__VERSION__"
          releaseDraft: true
          args: ${{ matrix.args }}
```

|OS|산출물|React 관련 체크|
|---|---|---|
|🪟 Windows|`.msi`, `-setup.exe`|WebView2(Chromium): 대부분 최신 기능 지원|
|🍎 macOS|`.dmg`, `.app` (universal)|WebKit: Blob 워커·CSS 호환성 점검, `⌘` 단축키|
|🐧 Ubuntu|`.deb`, `.AppImage`, `.rpm`|WebKitGTK: 폰트(한글 폰트 설치 여부), GPU 렌더링 차이|

<aside>  
💡

자동 업데이트와 코드 서명은 심화 과정 A의 T6과 동일하게 적용합니다(`tauri-plugin-updater` + 서명 키). 한글이 깨지지 않도록 CSS에 `font-family: system-ui, "Apple SD Gothic Neo", "Malgun Gothic", "Noto Sans KR", sans-serif;` 처럼 OS별 한글 폰트를 나열하세요.

</aside>
