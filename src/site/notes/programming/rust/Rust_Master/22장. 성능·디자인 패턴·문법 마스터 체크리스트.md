```rust
// benches/parse.rs
use criterion::{criterion_group, criterion_main, Criterion};
use std::hint::black_box;

fn sum_loop(v: &[u64]) -> u64 { let mut s = 0; for i in 0..v.len() { s += v[i]; } s }
fn sum_iter(v: &[u64]) -> u64 { v.iter().sum() }

fn bench(c: &mut Criterion) {
    let data: Vec<u64> = (0..100_000).collect();
    c.bench_function("index loop", |b| b.iter(|| sum_loop(black_box(&data))));
    c.bench_function("iterator", |b| b.iter(|| sum_iter(black_box(&data))));
}
criterion_group!(benches, bench);
criterion_main!(benches);
```

```bash
cargo bench
```

## 💡 해결 2 — 자주 만나는 성능 개선 포인트

|증상|개선|
|---|---|
|루프 안의 `.clone()`, `to_string()`|참조로 빌려 쓰기, `&str` 매개변수|
|`Vec` 에 반복 `push`|`Vec::with_capacity(n)` 로 미리 할당|
|문자열 `+` 반복|`String::with_capacity` • `push_str`, 또는 `write!`|
|가끔만 수정하는 문자열|`Cow<str>` 로 필요할 때만 복사|
|파일을 한 줄씩 `read`|`BufReader` / `BufWriter`|
|`Box<dyn Trait>` 가 핫패스에|제네릭·열거형으로 정적 디스패치|
|CPU 무거운 반복|rayon `par_iter()`|

```rust
use std::borrow::Cow;

// 공백이 없으면 복사 없이 그대로, 있을 때만 새 String 할당
fn normalize(input: &str) -> Cow<'_, str> {
    if input.contains(' ') {
        Cow::Owned(input.replace(' ', "_"))
    } else {
        Cow::Borrowed(input)
    }
}
```

## 💡 해결 3 — 관용구(Idioms) 모음

### 🏗️ 패턴 — 타입으로 불가능한 상태를 표현 불가능하게

```rust
// ❌ bool 두 개로 4가지 조합 (그중 하나는 불가능한 상태)
struct ConnBad { connected: bool, authenticated: bool }

// ✅ 열거형으로 가능한 상태만
enum Conn { Disconnected, Connected { session: u32 }, Authenticated { session: u32, user: String } }
```

### 🔄 변환 — From/TryFrom/AsRef를 구현하기

```rust
fn open<P: AsRef<std::path::Path>>(p: P) { let _ = p.as_ref(); }
open("a.txt"); open(String::from("b.txt")); open(std::path::PathBuf::from("c"));

struct Email(String);
impl TryFrom<&str> for Email {
    type Error = &'static str;
    fn try_from(s: &str) -> Result<Self, Self::Error> {
        if s.contains('@') { Ok(Email(s.into())) } else { Err("이메일 형식 아님") }
    }
}
```
