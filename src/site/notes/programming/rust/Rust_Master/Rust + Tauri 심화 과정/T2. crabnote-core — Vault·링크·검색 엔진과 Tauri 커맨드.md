## 🧩 문제 1 — 프론트엔드가 보낸 경로를 믿을 수 없다

## 💡 해결 1 — 상대 경로만 받고 Vault 안으로 가둠다

```rust
// crates/crabnote-core/src/lib.rs
pub mod links;

pub use vault::{NoteMeta, Vault};

#[derive(Debug, thiserror::Error)]
pub enum CoreError {
    #[error("파일 오류: {0}")]
    Io(#[from] std::io::Error),
    #[error("Vault 밖의 경로는 허용되지 않습니다: {0}")]
    OutsideVault(String),
    #[error("노트를 찾을 수 없습니다: {0}")]
    NotFound(String),
    #[error("이미 존재합니다: {0}")]
    AlreadyExists(String),
}

pub type Result<T> = std::result::Result<T, CoreError>;
```

```rust
// crates/crabnote-core/src/vault.rs
use crate::{links, CoreError, Result};
use serde::Serialize;
use std::collections::BTreeMap;
use std::path::{Component, Path, PathBuf};
use walkdir::WalkDir;

#[derive(Debug, Clone, Serialize)]
pub struct NoteMeta {
    pub path: String,   // Vault 기준 상대 경로, 항상 '/' 구분자 (OS 독립)
    pub title: String,  // 파일 이름(확장자 제외) = 위키링크 이름
}

pub struct Vault {
    root: PathBuf,
    notes: BTreeMap<String, NoteMeta>,  // key: path
}

impl Vault {
    pub fn open(root: impl Into<PathBuf>) -> Result<Self> {
        let root: PathBuf = root.into();
        if !root.is_dir() { return Err(CoreError::NotFound(root.display().to_string())); }
        let mut v = Self { root, notes: BTreeMap::new() };
        v.rescan();
        Ok(v)
    }

    pub fn root(&self) -> &Path { &self.root }

    pub fn rescan(&mut self) {
        self.notes = WalkDir::new(&self.root)
            .into_iter()
            .filter_entry(|e| !e.file_name().to_string_lossy().starts_with('.')) // .obsidian, .git 제외
            .filter_map(|e| e.ok())
            .filter(|e| e.file_type().is_file() && e.path().extension().is_some_and(|x| x == "md"))
            .filter_map(|e| {
                let rel = e.path().strip_prefix(&self.root).ok()?;
                let path = to_slash(rel);
                let title = e.path().file_stem()?.to_string_lossy().into_owned();
                Some((path.clone(), NoteMeta { path, title }))
            })
            .collect();
    }

    pub fn notes(&self) -> impl Iterator<Item = &NoteMeta> { self.notes.values() }

    /// 상대 경로를 실제 경로로. 절대 경로·'..'·드라이브 문자는 거부 (경로 조작 방지)
    pub fn resolve(&self, rel: &str) -> Result<PathBuf> {
        let p = Path::new(rel);
        let bad = p.components().any(|c| matches!(c, Component::ParentDir | Component::RootDir | Component::Prefix(_)));
        if rel.is_empty() || bad { return Err(CoreError::OutsideVault(rel.into())); }
        Ok(self.root.join(p))
    }

    pub fn read(&self, rel: &str) -> Result<String> {
        Ok(std::fs::read_to_string(self.resolve(rel)?)?)
    }

    pub fn write(&mut self, rel: &str, text: &str) -> Result<()> {
        let full = self.resolve(rel)?;
        if let Some(dir) = full.parent() { std::fs::create_dir_all(dir)?; }
        // 원자적 저장: 임시 파일에 쓴 뒤 이름 변경 → 저장 중 전원이 꺼져도 파일이 깨지지 않음
        let tmp = full.with_extension("md.tmp");
        std::fs::write(&tmp, text)?;
        std::fs::rename(&tmp, &full)?;
        self.rescan();
        Ok(())
    }

    pub fn find_by_title(&self, title: &str) -> Option<&NoteMeta> {
        let t = title.trim();
        self.notes.values().find(|n| n.title.eq_ignore_ascii_case(t) || n.path.trim_end_matches(".md") == t)
    }

    pub fn create(&mut self, title: &str) -> Result<NoteMeta> {
        let safe: String = title.trim().chars().filter(|c| !r#"\/:*?"<>|"#.contains(*c)).collect();
        let rel = format!("{safe}.md");
        if self.notes.contains_key(&rel) { return Err(CoreError::AlreadyExists(rel)); }
        self.write(&rel, &format!("# {safe}\n\n"))?;
        Ok(self.notes[&rel].clone())
    }

    /// 이름 변경 + 모든 노트의 [[옛이름]] 을 [[새이름]] 으로 갱신 (Obsidian의 핵심 기능)
    pub fn rename(&mut self, rel: &str, new_title: &str) -> Result<(NoteMeta, usize)> {
        let old = self.notes.get(rel).cloned().ok_or_else(|| CoreError::NotFound(rel.into()))?;
        let new_rel = match rel.rsplit_once('/') {
            Some((dir, _)) => format!("{dir}/{new_title}.md"),
            None => format!("{new_title}.md"),
        };
        let target = self.resolve(&new_rel)?;
        if target.exists() { return Err(CoreError::AlreadyExists(new_rel)); }
        std::fs::rename(self.resolve(rel)?, &target)?;

        let mut updated = 0;
        let paths: Vec<String> = self.notes.keys().filter(|p| *p != rel).cloned().collect();
        for p in paths.iter().map(String::as_str).chain([new_rel.as_str()]) {
            let body = self.read(p)?;
            let replaced = links::rename_links(&body, &old.title, new_title);
            if replaced != body {
                std::fs::write(self.resolve(p)?, replaced)?;
                updated += 1;
            }
        }
        self.rescan();
        Ok((self.notes[&new_rel].clone(), updated))
    }
}

fn to_slash(p: &Path) -> String {
    p.components().map(|c| c.as_os_str().to_string_lossy()).collect::<Vec<_>>().join("/")
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn rejects_path_traversal() {
        let dir = tempfile::tempdir().unwrap();
        let v = Vault::open(dir.path()).unwrap();
        assert!(v.resolve("../secret.txt").is_err());
        assert!(v.resolve("/etc/passwd").is_err());
        assert!(v.resolve("notes/a.md").is_ok());
    }

    #[test]
    fn rename_updates_links() {
        let dir = tempfile::tempdir().unwrap();
        let mut v = Vault::open(dir.path()).unwrap();
        v.write("Rust.md", "# Rust").unwrap();
        v.write("Diary.md", "오늘 [[Rust]] 공부, [[rust|러스트]]").unwrap();
        let (_, n) = v.rename("Rust.md", "Rust언어").unwrap();
        assert_eq!(n, 1);
        assert_eq!(v.read("Diary.md").unwrap(), "오늘 [[Rust언어]] 공부, [[Rust언어|러스트]]");
    }
}
```

## 💡 해결 2 — `links.rs`

```rust
// crates/crabnote-core/src/links.rs
use regex::{Captures, Regex};
use std::collections::{BTreeSet, HashMap};
use std::sync::LazyLock;

// [[대상]] [[대상|별칭]] [[대상#제목]] [[대상#제목|별칭]]
pub static WIKILINK: LazyLock<Regex> = LazyLock::new(|| {
    Regex::new(r"\[\[(?<target>[^\]|#]+)(?<heading>#[^\]|]*)?(?:\|(?<alias>[^\]]*))?\]\]").unwrap()
});
static TAG: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?:^|\s)#(?<tag>[\p{L}\p{N}_/-]+)").unwrap());

pub fn outgoing(body: &str) -> BTreeSet<String> {
    WIKILINK.captures_iter(body).map(|c| c["target"].trim().to_string()).collect()
}

pub fn tags(body: &str) -> BTreeSet<String> {
    TAG.captures_iter(body).map(|c| c["tag"].to_string()).collect()
}

/// [[old]] → [[new]] (대소문자 무시, heading·alias 유지)
pub fn rename_links(body: &str, old: &str, new: &str) -> String {
    WIKILINK
        .replace_all(body, |c: &Captures| {
            if !c["target"].trim().eq_ignore_ascii_case(old) { return c[0].to_string(); }
            let heading = c.name("heading").map_or("", |m| m.as_str());
            match c.name("alias") {
                Some(a) => format!("[[{new}{heading}|{}]]", a.as_str()),
                None => format!("[[{new}{heading}]]"),
            }
        })
        .into_owned()
}

/// 제목(소문자) → 그 노트를 링크한 노트 경로들
#[derive(Default)]
pub struct LinkIndex {
    backlinks: HashMap<String, BTreeSet<String>>,
    tags: HashMap<String, BTreeSet<String>>,
}

impl LinkIndex {
    pub fn build<'a>(notes: impl Iterator<Item = (&'a str, String)>) -> Self {
        let mut idx = Self::default();
        for (path, body) in notes {
            for t in outgoing(&body) {
                idx.backlinks.entry(t.to_lowercase()).or_default().insert(path.to_string());
            }
            for tag in tags(&body) {
                idx.tags.entry(tag).or_default().insert(path.to_string());
            }
        }
        idx
    }
    pub fn backlinks(&self, title: &str) -> Vec<String> {
        self.backlinks.get(&title.to_lowercase()).map(|s| s.iter().cloned().collect()).unwrap_or_default()
    }
    pub fn notes_with_tag(&self, tag: &str) -> Vec<String> {
        self.tags.get(tag).map(|s| s.iter().cloned().collect()).unwrap_or_default()
    }
    pub fn all_tags(&self) -> Vec<(String, usize)> {
        let mut v: Vec<_> = self.tags.iter().map(|(k, s)| (k.clone(), s.len())).collect();
        v.sort_by(|a, b| b.1.cmp(&a.1).then(a.0.cmp(&b.0)));
        v
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    #[test] fn parses_all_forms() {
        let s = outgoing("[[A]] [[B|비]] [[C#제목]] [[D#h|디]]");
        assert_eq!(s.into_iter().collect::<Vec<_>>(), ["A", "B", "C", "D"]);
    }
    #[test] fn finds_korean_tags() {
        assert!(tags("회의 #프로젝트/크랩 메모").contains("프로젝트/크랩"));
    }
}
```

## 🧩 문제 3 — 수천 개 노트에서 빠르게 찾고, 미리보기를 안전하게 만들어야 한다

## 💡 해결 3 — `search.rs`(rayon) + `render.rs`(pulldown-cmark + ammonia)

```rust
// crates/crabnote-core/src/search.rs
use crate::Vault;
use rayon::prelude::*;
use serde::Serialize;

#[derive(Debug, Serialize)]
pub struct Hit { pub path: String, pub title: String, pub line: usize, pub snippet: String }

pub fn search(vault: &Vault, query: &str, limit: usize) -> Vec<Hit> {
    let q = query.to_lowercase();
    if q.is_empty() { return vec![]; }
    let notes: Vec<_> = vault.notes().cloned().collect();
    let mut hits: Vec<Hit> = notes
        .par_iter()                                   // 노트별 병렬 검색 (L9 rayon)
        .flat_map_iter(|n| {
            let body = vault.read(&n.path).unwrap_or_default();
            body.lines()
                .enumerate()
                .filter(|(_, l)| l.to_lowercase().contains(&q))
                .map(|(i, l)| Hit { path: n.path.clone(), title: n.title.clone(), line: i + 1,
                                    snippet: l.chars().take(120).collect() })
                .collect::<Vec<_>>()
        })
        .collect();
    hits.sort_by(|a, b| a.path.cmp(&b.path).then(a.line.cmp(&b.line)));
    hits.truncate(limit);
    hits
}
```

```rust
// crates/crabnote-core/src/render.rs
use crate::links::WIKILINK;
use pulldown_cmark::{html, Options, Parser};

fn escape(s: &str) -> String {
    s.replace('&', "&amp;").replace('<', "&lt;").replace('>', "&gt;").replace('"', "&quot;")
}

/// 마크다운 → 안전한 HTML. 위키링크는 클릭 가능한 <a class="wikilink" data-target="...">로 변환
pub fn to_html(md: &str) -> String {
    let with_links = WIKILINK.replace_all(md, |c: &regex::Captures| {
        let target = c["target"].trim();
        let shown = c.name("alias").map_or(target, |m| m.as_str());
        format!(r#"<a class="wikilink" data-target="{}">{}</a>"#, escape(target), escape(shown))
    });
    let mut raw = String::new();
    html::push_html(&mut raw, Parser::new_ext(&with_links, Options::all()));

    // 🔒 XSS 방어: script, on* 이벤트 속성, javascript: URL 등을 제거하고 필요한 속성만 허용
    ammonia::Builder::default()
        .add_generic_attributes(["class", "data-target"])
        .clean(&raw)
        .to_string()
}

#[cfg(test)]
mod tests {
    #[test]
    fn strips_scripts_keeps_wikilinks() {
        let html = super::to_html("[[노트|보기]] <script>alert(1)</script> <img src=x onerror=alert(1)>");
        assert!(html.contains(r#"data-target="노트""#));
        assert!(!html.contains("<script") && !html.contains("onerror"));
    }
}
```

## 🧩 문제 4 — 이 엔진을 프론트엔드에 노출하고, 외부 변경도 알려줘야 한다

## 💡 해결 4 — Tauri 커맨드 + 상태 + notify 감시자

```rust
// src-tauri/src/lib.rs
mod plugins;   // T5

use crabnote_core::{links::LinkIndex, render, search, NoteMeta, Vault};
use serde::Serialize;
use std::sync::Mutex;
use tauri::{AppHandle, Emitter, Manager, State};

#[derive(Default)]
pub struct AppState {
    vault: Mutex<Option<Vault>>,
    index: Mutex<LinkIndex>,
    watcher: Mutex<Option<notify::RecommendedWatcher>>, // 드롭되면 감시가 멈추므로 보관
    plugins: Mutex<plugins::PluginHost>,
}

type CmdResult<T> = Result<T, String>;   // 프론트엔드에는 사람이 읽을 수 있는 메시지로

fn with_vault<T>(s: &AppState, f: impl FnOnce(&mut Vault) -> crabnote_core::Result<T>) -> CmdResult<T> {
    let mut guard = s.vault.lock().unwrap();
    let v = guard.as_mut().ok_or("Vault가 열려 있지 않습니다")?;
    f(v).map_err(|e| e.to_string())
}

fn rebuild_index(s: &AppState) {
    if let Some(v) = s.vault.lock().unwrap().as_ref() {
        let pairs = v.notes().filter_map(|n| v.read(&n.path).ok().map(|b| (n.path.as_str(), b)));
        *s.index.lock().unwrap() = LinkIndex::build(pairs);
    }
}

#[tauri::command]
fn open_vault(app: AppHandle, path: String, state: State<'_, AppState>) -> CmdResult<Vec<NoteMeta>> {
    let vault = Vault::open(&path).map_err(|e| e.to_string())?;
    let notes = vault.notes().cloned().collect();

    // 외부 편집기·동기화 도구가 파일을 바꾸면 프론트엔드에 이벤트로 알림
    use notify::Watcher;
    let handle = app.clone();
    let mut watcher = notify::recommended_watcher(move |res: notify::Result<notify::Event>| {
        if let Ok(ev) = res {
            let changed: Vec<String> = ev.paths.iter().map(|p| p.display().to_string()).collect();
            let _ = handle.emit("vault-changed", changed);
        }
    })
    .map_err(|e| e.to_string())?;
    watcher.watch(vault.root(), notify::RecursiveMode::Recursive).map_err(|e| e.to_string())?;

    *state.watcher.lock().unwrap() = Some(watcher);
    *state.vault.lock().unwrap() = Some(vault);
    rebuild_index(&state);
    Ok(notes)
}

#[tauri::command]
fn list_notes(state: State<'_, AppState>) -> CmdResult<Vec<NoteMeta>> {
    with_vault(&state, |v| { v.rescan(); Ok(v.notes().cloned().collect()) })
}

#[tauri::command]
fn read_note(path: String, state: State<'_, AppState>) -> CmdResult<String> {
    with_vault(&state, |v| v.read(&path))
}

#[derive(Serialize)]
struct SaveResult { text: String, notices: Vec<String> }

#[tauri::command]
fn save_note(path: String, text: String, state: State<'_, AppState>) -> CmdResult<SaveResult> {
    // T5: 저장 전에 플러그인들의 on_save 훅을 순서대로 적용
    let title = path.trim_end_matches(".md").rsplit('/').next().unwrap_or(&path).to_string();
    let (text, notices) = state.plugins.lock().unwrap().run_on_save(&path, &title, text);
    with_vault(&state, |v| v.write(&path, &text))?;
    rebuild_index(&state);
    Ok(SaveResult { text, notices })
}

#[tauri::command]
fn open_link(title: String, state: State<'_, AppState>) -> CmdResult<NoteMeta> {
    let r = with_vault(&state, |v| match v.find_by_title(&title) {
        Some(n) => Ok(n.clone()),
        None => v.create(&title),                 // 없는 링크 클릭 → 새 노트 (Obsidian 동작)
    });
    rebuild_index(&state);
    r
}

#[tauri::command]
fn rename_note(path: String, new_title: String, state: State<'_, AppState>) -> CmdResult<(NoteMeta, usize)> {
    let r = with_vault(&state, |v| v.rename(&path, &new_title));
    rebuild_index(&state);
    r
}

#[tauri::command]
fn backlinks(title: String, state: State<'_, AppState>) -> Vec<String> {
    state.index.lock().unwrap().backlinks(&title)
}

#[tauri::command]
fn all_tags(state: State<'_, AppState>) -> Vec<(String, usize)> {
    state.index.lock().unwrap().all_tags()
}

#[tauri::command]
fn search_notes(query: String, state: State<'_, AppState>) -> CmdResult<Vec<search::Hit>> {
    with_vault(&state, |v| Ok(search::search(v, &query, 200)))
}

#[tauri::command]
fn render_markdown(text: String) -> String {
    render::to_html(&text)
}

#[cfg_attr(mobile, tauri::mobile_entry_point)]
pub fn run() {
    tauri::Builder::default()
        .plugin(tauri_plugin_dialog::init())
        .manage(AppState::default())
        .setup(|app| {
            // T5: <앱 데이터>/plugins 에서 서드파티 플러그인 로드
            let dir = app.path().app_data_dir()?.join("plugins");
            std::fs::create_dir_all(&dir)?;
            let (host, errors) = plugins::PluginHost::load_all(&dir);
            for e in errors { eprintln!("⚠️ 플러그인 로드 실패: {e}"); }
            *app.state::<AppState>().plugins.lock().unwrap() = host;
            Ok(())
        })
        .invoke_handler(tauri::generate_handler![
            open_vault, list_notes, read_note, save_note, open_link, rename_note,
            backlinks, all_tags, search_notes, render_markdown,
            plugins::list_plugins, plugins::plugin_commands, plugins::run_plugin_command,
            plugins::set_plugin_enabled, plugins::reload_plugins, plugins::plugins_dir,
        ])
        .run(tauri::generate_context!())
        .expect("CrabNote Pro 실행 실패");
}
```

<aside>  
⚠️

**함정 — 락 순서와 재진입**: `with_vault` 안에서 `rebuild_index` 를 부르면 `vault` 락을 두 번 잡아 **데드락**이 납니다(std Mutex는 재진입 불가). 그래서 위 코드는 항상 `with_vault` 가 끝난 뒤에 인덱스를 다시 만듭니다. 대규모 Vault에서는 인덱스를 전체 재구축하지 말고 바뀐 노트만 갱신하도록 개선하세요(도전 과제).

</aside>

📌

`notify` 는 한 번 저장에도 여러 이벤트를 보낼 수 있습니다. 실제 제품에서는 `notify-debouncer-full` 로 묶거나 프론트엔드에서 300ms 디바운스하세요(T3에서 적용). 또 앱 자신의 저장도 이벤트를 발생시키므로, 편집 중인 노트를 덮어쓰지 않도록 dirty 상태를 확인해야 합니다.
