## 🧩 문제 1 — Rust 커맨드를 타입 없이 부르면 오타가 런타임에야 드러난다

## 💡 해결 1 — 타입 있는 API 래퍼 한 곳에 모으기

```tsx
// src/api.ts
import { invoke } from "@tauri-apps/api/core";

export type NoteMeta = { path: string; title: string };
export type Hit = { path: string; title: string; line: number; snippet: string };
export type PluginCommand = { plugin_id: string; id: string; name: string };

export const api = {
  openVault: (path: string) => invoke<NoteMeta[]>("open_vault", { path }),
  listNotes: () => invoke<NoteMeta[]>("list_notes"),
  read: (path: string) => invoke<string>("read_note", { path }),
  save: (path: string, text: string) => invoke<{ text: string; notices: string[] }>("save_note", { path, text }),
  openLink: (title: string) => invoke<NoteMeta>("open_link", { title }),
  rename: (path: string, newTitle: string) => invoke<[NoteMeta, number]>("rename_note", { path, newTitle }),
  backlinks: (title: string) => invoke<string[]>("backlinks", { title }),
  tags: () => invoke<[string, number][]>("all_tags"),
  search: (query: string) => invoke<Hit[]>("search_notes", { query }),
  render: (text: string) => invoke<string>("render_markdown", { text }),
  pluginCommands: () => invoke<PluginCommand[]>("plugin_commands"),
  runPluginCommand: (pluginId: string, commandId: string, path: string, text: string, selection: string) =>
    invoke<{ text?: string; insert?: string; notice?: string }>("run_plugin_command", { pluginId, commandId, path, text, selection }),
};
```

🔍

Tauri는 Rust 인자 `new_title` 을 JS에서 `newTitle`(camelCase)로 받습니다. 타입을 Rust에서 자동 생성하고 싶다면 `tauri-specta` 크레이트를 검토하세요.

## 🧩 문제 2 — 에디터·미리보기·자동 저장·`[[` 자동완성

## 💡 해결 2 — CodeMirror 6 + 디바운스 + 자동완성 소스

```tsx
// src/main.ts
import { EditorView, basicSetup } from "codemirror";
import { keymap } from "@codemirror/view";
import { markdown } from "@codemirror/lang-markdown";
import { autocompletion, CompletionContext } from "@codemirror/autocomplete";
import { listen } from "@tauri-apps/api/event";
import { open } from "@tauri-apps/plugin-dialog";
import { api, NoteMeta } from "./api";
import { setupPalette } from "./palette";
import "./styles.css";

const $ = <T extends HTMLElement>(sel: string) => document.querySelector<T>(sel)!;

let notes: NoteMeta[] = [];
let current: NoteMeta | null = null;
let dirty = false;
let saveTimer: number | undefined;
let previewTimer: number | undefined;

const status = (msg: string) => ($("#status").textContent = msg);

// --- [[ 자동완성: [[ 뒤에서 노트 제목 제안 ---
function wikiCompletion(ctx: CompletionContext) {
  const m = ctx.matchBefore(/\[\[[^\]|#]*/);
  if (!m) return null;
  return {
    from: m.from + 2,
    options: notes.map((n) => ({ label: n.title, detail: n.path, apply: `${n.title}]]` })),
  };
}

const view = new EditorView({
  parent: $("#editor"),
  extensions: [
    basicSetup,
    markdown(),
    EditorView.lineWrapping,
    autocompletion({ override: [wikiCompletion] }),
    keymap.of([{ key: "Mod-s", run: () => { void save(); return true; } }]),
    EditorView.updateListener.of((u) => {
      if (!u.docChanged) return;
      dirty = true;
      clearTimeout(previewTimer);
      previewTimer = window.setTimeout(renderPreview, 150);  // 입력 중 과도한 렌더링 방지
      clearTimeout(saveTimer);
      saveTimer = window.setTimeout(save, 1500);              // 1.5초 멈추면 자동 저장
    }),
  ],
});

const text = () => view.state.doc.toString();
function setText(t: string) {
  view.dispatch({ changes: { from: 0, to: view.state.doc.length, insert: t } });
  dirty = false;
}

async function renderPreview() {
  $("#preview").innerHTML = await api.render(text());   // Rust에서 ammonia로 정화된 HTML
}

async function save() {
  if (!current || !dirty) return;
  const res = await api.save(current.path, text());
  if (res.text !== text()) setText(res.text);               // 플러그인이 변환한 결과 반영
  dirty = false;
  status(res.notices.length ? `💾 저장 · ${res.notices.join(" · ")}` : `💾 ${current.title} 저장됨`);
  await refreshSide();
}

export async function openNote(n: NoteMeta) {
  await save();                                             // 떠나기 전 저장
  current = n;
  setText(await api.read(n.path));
  await renderPreview();
  await refreshSide();
  renderFiles();
  status(n.path);
}

export async function openByTitle(title: string) {
  const n = await api.openLink(title);
  notes = await api.listNotes();
  await openNote(n);
}

function renderFiles() {
  const q = $<HTMLInputElement>("#filter").value.toLowerCase();
  const ul = $("#files");
  ul.replaceChildren(
    ...notes.filter((n) => n.path.toLowerCase().includes(q)).map((n) => {
      const li = document.createElement("li");
      li.textContent = `📄 ${n.title}`;                     // textContent → 파일명으로 XSS 불가
      li.title = n.path;
      li.classList.toggle("active", n.path === current?.path);
      li.onclick = () => openNote(n);
      return li;
    }),
  );
}

async function refreshSide() {
  if (!current) return;
  const bl = await api.backlinks(current.title);
  $("#backlinks").replaceChildren(...bl.map((p) => {
    const li = document.createElement("li");
    li.textContent = p;
    li.onclick = () => { const n = notes.find((x) => x.path === p); if (n) openNote(n); };
    return li;
  }));
  const tags = await api.tags();
  $("#tags").replaceChildren(...tags.slice(0, 30).map(([t, c]) => {
    const li = document.createElement("li");
    li.textContent = `#${t} (${c})`;
    return li;
  }));
}

// 미리보기의 위키링크 클릭 → 이동 (이벤트 위임)
$("#preview").addEventListener("click", (e) => {
  const a = (e.target as HTMLElement).closest<HTMLAnchorElement>("a.wikilink");
  if (a?.dataset.target) { e.preventDefault(); void openByTitle(a.dataset.target); }
});

$("#filter").addEventListener("input", renderFiles);

$("#open-vault").addEventListener("click", async () => {
  const dir = await open({ directory: true, title: "Vault 폴더 선택" });
  if (typeof dir !== "string") return;
  notes = await api.openVault(dir);
  renderFiles();
  status(`${notes.length}개 노트`);
});

// 외부에서 파일이 바뀌면 목록 갱신 (디바운스), 편집 중이 아니면 현재 노트도 다시 읽기
let fsTimer: number | undefined;
void listen<string[]>("vault-changed", () => {
  clearTimeout(fsTimer);
  fsTimer = window.setTimeout(async () => {
    notes = await api.listNotes();
    renderFiles();
    if (current && !dirty) {
      const latest = await api.read(current.path).catch(() => null);
      if (latest !== null && latest !== text()) { setText(latest); await renderPreview(); }
    }
  }, 300);
});

setupPalette({ getNotes: () => notes, openNote, getCurrent: () => current, getText: text, setText, view, status });
```

## 🧩 문제 3 — 노트 전환·검색·플러그인 명령을 키보드로 하고 싶다

## 💡 해결 3 — 통합 팔레트 (퀵 스위처 + 검색 + 명령)

```tsx
// src/palette.ts
import type { EditorView } from "codemirror";
import { api, NoteMeta } from "./api";

type Item = { label: string; hint?: string; run: () => void | Promise<void> };
type Ctx = {
  getNotes: () => NoteMeta[]; openNote: (n: NoteMeta) => Promise<void>; getCurrent: () => NoteMeta | null;
  getText: () => string; setText: (t: string) => void; view: EditorView; status: (m: string) => void;
};

export function setupPalette(ctx: Ctx) {
  const dlg = document.querySelector<HTMLDialogElement>("#palette")!;
  const input = document.querySelector<HTMLInputElement>("#palette-input")!;
  const list = document.querySelector<HTMLUListElement>("#palette-list")!;
  let mode: "notes" | "commands" = "notes";
  let items: Item[] = [];
  let sel = 0;

  async function commandItems(): Promise<Item[]> {
    const builtins: Item[] = [
      { label: "노트 이름 바꾸기", run: async () => {
          const cur = ctx.getCurrent(); if (!cur) return;
          const name = prompt("새 이름", cur.title); if (!name) return;
          const [meta, n] = await api.rename(cur.path, name);
          ctx.status(`✏️ ${n}개 노트의 링크를 갱신했습니다`);
          await ctx.openNote(meta);
      } },
    ];
    // 🧩 T5: 플러그인이 등록한 명령
    const pluginCmds = (await api.pluginCommands()).map<Item>((c) => ({
      label: c.name, hint: `🧩 ${c.plugin_id}`,
      run: async () => {
        const cur = ctx.getCurrent(); if (!cur) return;
        const { state } = ctx.view;
        const selection = state.sliceDoc(state.selection.main.from, state.selection.main.to);
        const r = await api.runPluginCommand(c.plugin_id, c.id, cur.path, ctx.getText(), selection);
        if (r.text !== undefined && r.text !== null) ctx.setText(r.text);
        if (r.insert) ctx.view.dispatch(ctx.view.state.replaceSelection(r.insert)); // 최신 상태 기준
        if (r.notice) ctx.status(`🧩 ${r.notice}`);
      },
    }));
    return [...builtins, ...pluginCmds];
  }

  async function refresh() {
    const q = input.value.trim().toLowerCase();
    if (mode === "commands") {
      items = (await commandItems()).filter((i) => i.label.toLowerCase().includes(q));
    } else if (q.startsWith("/")) {                         // "/키워드" → 전문 검색
      const hits = await api.search(q.slice(1));
      items = hits.map((h) => ({ label: `${h.title}:${h.line}`, hint: h.snippet,
        run: () => { const n = ctx.getNotes().find((x) => x.path === h.path); if (n) return ctx.openNote(n); } }));
    } else {
      items = ctx.getNotes().filter((n) => n.title.toLowerCase().includes(q)).slice(0, 50)
        .map((n) => ({ label: n.title, hint: n.path, run: () => ctx.openNote(n) }));
    }
    sel = 0;
    draw();
  }

  function draw() {
    list.replaceChildren(...items.map((it, i) => {
      const li = document.createElement("li");
      li.textContent = it.hint ? `${it.label} — ${it.hint}` : it.label;
      li.className = i === sel ? "active" : "";
      li.onclick = () => choose(i);
      return li;
    }));
  }

  async function choose(i: number) { dlg.close(); await items[i]?.run(); }

  function show(m: typeof mode) { mode = m; input.value = ""; dlg.showModal(); input.focus(); void refresh(); }

  input.addEventListener("input", () => void refresh());
  input.addEventListener("keydown", (e) => {
    if (e.key === "ArrowDown") { sel = Math.min(sel + 1, items.length - 1); draw(); e.preventDefault(); }
    if (e.key === "ArrowUp") { sel = Math.max(sel - 1, 0); draw(); e.preventDefault(); }
    if (e.key === "Enter") void choose(sel);
  });
  window.addEventListener("keydown", (e) => {
    const mod = e.ctrlKey || e.metaKey;                  // Windows/Linux: Ctrl, macOS: Cmd
    if (mod && e.key.toLowerCase() === "o") { e.preventDefault(); show("notes"); }
    if (mod && e.key.toLowerCase() === "p") { e.preventDefault(); show("commands"); }
  });
  document.querySelector("#palette-btn")!.addEventListener("click", () => show("commands"));
}
```
